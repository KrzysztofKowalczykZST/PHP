function KALKULATOR($a, $b, $dzialanie) {
    $wynik = 0;
    switch ($dzialanie) {
        case 'suma':
            $wynik = $a + $b;
            break;
        case 'roznica':
            $wynik = $a - $b;
            break;
        case 'iloczyn':
            $wynik = $a * $b;
            break;
        case 'iloraz':
            $wynik = ($b != 0) ? ($a / $b) : "Błąd: dzielenie przez 0!";
            break;
        default:
            $wynik = "Nieznane działanie";
    }
    
    echo "<div id='wynik'>$wynik</div>";

<?php
function BMI($wzrost_cm, $waga_kg) {
    $wzrost_m = $wzrost_cm / 100;
    if ($wzrost_m <= 0) {
        echo "<div id='wynik'>Błędny wzrost</div>";
        return;
    }
    
    $bmi = $waga_kg / ($wzrost_m * $wzrost_m);
    $bmi_format = round($bmi, 2);
    $komentarz = "";

    if ($bmi < 18.5) {
        $komentarz = "za mało!";
    } elseif ($bmi > 25) {
        $komentarz = "za dużo!";
    } else {
        $komentarz = "OK!";
    }

    echo "<div id='wynik'>BMI: $bmi_format - $komentarz</div>";
}

<?php
function SIŁA($haslo) {
    $dlugosc = strlen($haslo);

    if ($dlugosc <= 4) {
        echo "hasło słabe";
        return;
    }
    if (!preg_match('/[0-9]/', $haslo)) {
        echo "hasło słabe";
        return;
    }
    if (!preg_match('/[A-Z]/', $haslo)) {
        echo "hasło słabe";
        return;
    }
    if (!preg_match('/[a-z]/', $haslo)) {
        echo "hasło słabe";
        return;
    }
    if (!preg_match('/[^a-zA-Z0-9]/', $haslo)) {
        echo "hasło słabe";
        return;
    }

    if ($dlugosc < 8) {
        echo "hasło średnie";
    } else {
        echo "hasło mocne";
    }
}

<?php
function SZYFR($tekst) {
    $zaszyfrowany = "";
    $dlugosc = strlen($tekst);

    for ($i = 0; $i < $dlugosc; $i++) {
        $char = $tekst[$i];
        
        if ($char >= 'a' && $char <= 'z') {
            // Przesunięcie dla małych liter z zapętleniem
            $zaszyfrowany .= chr(((ord($char) - ord('a') + 2) % 26) + ord('a'));
        } elseif ($char >= 'A' && $char <= 'Z') {
            // Przesunięcie dla wielkich liter z zapętleniem
            $zaszyfrowany .= chr(((ord($char) - ord('A') + 2) % 26) + ord('A'));
        } else {
            // Pozostałe znaki (np. spacje, cyfry) zostają bez zmian
            $zaszyfrowany .= $char;
        }
    }

    echo "<input type='text' value='" . htmlspecialchars($tekst) . "' style='display:inline-block;'>";
    echo "<span style='margin-left: 10px; font-weight: bold;'>$zaszyfrowany</span>";
}
